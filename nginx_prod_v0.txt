server {
    server_name insighteye.io www.insighteye.io;
    
    # Increase header buffer size
    large_client_header_buffers 4 16k;
    
    # Root location for React frontend
    location / {
        proxy_pass http://127.0.0.1:8080;
        
        # Proxy headers
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # WebSocket support
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        
        # Timeouts for long-running requests
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
        
        # SPA routing: Try to serve requested URI first, then fall back to index.html
        try_files $uri $uri/ /index.html;
    }
    
    # API endpoint - FastAPI backend
    location /insighteye/ {
        proxy_pass http://127.0.0.1:8000/;
        
        # Proxy headers
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # WebSocket support
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        
        # Timeouts for long-running requests
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }
    
    # Static files handling with caching
    location ~* \.(js|css|png|jpg|jpeg|gif|ico)$ {
        expires max;
        log_not_found off;
        
        # Change this to proxy to your React container
        proxy_pass http://127.0.0.1:8080;
    }
    
    # Increase the maximum allowed size for requests
    client_max_body_size 100M;

    # Add error logging
    error_log /var/log/nginx/insighteye_error.log;
    access_log /var/log/nginx/insighteye_access.log;

    listen 443 ssl; # managed by Certbot
    #ssl_certificate /etc/letsencrypt/live/insighteye.io/fullchain.pem; # managed by Certbot
    #ssl_certificate_key /etc/letsencrypt/live/insighteye.io/privkey.pem; # managed by Certbot

    ssl_certificate /etc/nginx/ssl/insighteye.io/fullchain.pem; # copied from Let's Encrypt
    ssl_certificate_key /etc/nginx/ssl/insighteye.io/privkey.pem; # copied from Let's Encrypt

    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot
}

# HTTP redirect to HTTPS
server {
    listen 80;
    server_name insighteye.io www.insighteye.io;
    return 301 https://$host$request_uri;
}
