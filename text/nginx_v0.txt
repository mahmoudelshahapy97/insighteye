server {
    server_name insighteye.io www.insighteye.io;
    
    # Increase header buffer size
    large_client_header_buffers 4 16k;
    
    location / {
        # Update this to the Docker container's internal IP and port
        # If you're using Docker, you should use the container's name instead of localhost
        proxy_pass http://127.0.0.1:8080;
        
        # Proxy headers
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # WebSocket support
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        
        # Timeouts for long-running requests
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }
    
    # Increase the maximum allowed size for requests
    client_max_body_size 100M;

    # Add this to serve static files efficiently
    location ~* \.(js|css|png|jpg|jpeg|gif|ico)$ {
        expires max;
        log_not_found off;
        
        # Change this to proxy to your Docker container too
        proxy_pass http://127.0.0.1:8080;
    }

    # Add error logging
    error_log /var/log/nginx/insighteye_error.log;
    access_log /var/log/nginx/insighteye_access.log;

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/insighteye.io/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/insighteye.io/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}
